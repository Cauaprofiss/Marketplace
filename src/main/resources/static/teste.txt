### 1. Pacote `usuarios`

#### `Usuario.java` (Herança com `@Inheritance`)

> No Spring JPA, usamos `@Inheritance(strategy = InheritanceType.JOINED)` para herança. O banco de dados criará tabelas separadas para Aluno e Administrador, conectando-as pelo ID do Usuário.

```java
package br.com.sistema.model.usuarios;

import jakarta.persistence.*;
import java.util.Date;

@Entity
@Table(name = "usuarios")
@Inheritance(strategy = InheritanceType.JOINED) // Estrutura de herança no BD
public abstract class Usuario {
    
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    protected Long id;

    @Column(nullable = false, unique = true)
    protected String email;

    @Column(nullable = false)
    protected String senha;

    @Temporal(TemporalType.DATE)
    protected Date dataCadastro;

    protected String statusDisponibilidade;
    protected String localizacaoDeVenda;

    // Getters, Setters, Construtores...
}
```

```java
    // Aluno 0..* <-> 0..* Arrecadação (Muitos para Muitos precisa de uma tabela intermediária)
    @ManyToMany(mappedBy = "alunosParticipantes")
```

```java
package br.com.sistema.model.comercial;

import br.com.sistema.model.usuarios.Aluno;
import br.com.sistema.model.anuncios.Anuncio;
import br.com.sistema.model.financeiro.Pagamento;
import br.com.sistema.model.interacoes.MensagemChat;
import br.com.sistema.model.interacoes.Avaliacao;
import jakarta.persistence.*;
import java.util.Date;
import java.util.List;
import java.util.ArrayList;

@Entity
@Table(name = "pedidos")
public class Pedido {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Temporal(TemporalType.TIMESTAMP)
    private Date dataPedido;

    private String statusPedido;
    private String horarioEntregaAgendado;

    @Temporal(TemporalType.DATE)
    private Date dataEntregaAgendada;

    // Pedido * -> 1 Aluno (Muitos pedidos para 1 Aluno)
    @ManyToOne
    @JoinColumn(name = "aluno_id", nullable = false)
    private Aluno cliente;

    // Pedido * -> 1 Anuncio
    @ManyToOne
    @JoinColumn(name = "anuncio_id", nullable = false)
    private Anuncio anuncio;

    // Pedido 1 -> 0..* MensagemChat
    @OneToMany(mappedBy = "pedidoContexto", cascade = CascadeType.ALL)
    private List<MensagemChat> chat = new ArrayList<>();

    // Pedido 1 -> 0..* Pagamento
    @OneToMany(mappedBy = "pedido", cascade = CascadeType.ALL)
    private List<Pagamento> pagamentos = new ArrayList<>();

    // Pedido 1 -> 0..* Avaliacao
    @OneToMany(mappedBy = "pedidoOrigem", cascade = CascadeType.ALL)
    private List<Avaliacao> avaliacoes = new ArrayList<>();

    // Métodos de negócio...
}
```

```java
package br.com.sistema.model.comercial;

import br.com.sistema.model.usuarios.Aluno;
import br.com.sistema.model.interacoes.Avaliacao;
import jakarta.persistence.*;
import java.util.Date;
import java.util.List;
import java.util.ArrayList;

@Entity
@Table(name = "arrecadacoes")
public class Arrecadacao {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private String nomeTurma;
    private String descricao;
    private float objetivoFinanceiro;
    private float valorArrecadado;

    @Temporal(TemporalType.DATE)
    private Date dataLimite;

    // Arrecadação 0..* <-> 0..* Aluno (Criando a tabela de junção automática no BD)
    @ManyToMany
    @JoinTable(
        name = "aluno_arrecadacao",
        joinColumns = @JoinColumn(name = "arrecadacao_id"),
        inverseJoinColumns = @JoinColumn(name = "aluno_id")
    )
    private List<Aluno> alunosParticipantes = new ArrayList<>();

    // Arrecadação 0..1 -> 0..* Avaliação
    @OneToMany(mappedBy = "arrecadacaoReferente")
    private List<Avaliacao> avaliacoes = new ArrayList<>();

    // Métodos de negócio...
}
```

```java
package br.com.sistema.model.interacoes;

import br.com.sistema.model.usuarios.Aluno;

import java.util.List;
import java.util.ArrayList;

@Table(name = "avaliacoes")

    private String comentario;

    @Temporal(TemporalType.TIMESTAMP)

    @ManyToOne
    @JoinColumn(name = "pedido_id", nullable = false)
    private Pedido pedidoOrigem;

    // Avaliacao 0..* -> 0..1 Arrecadação
    @ManyToOne
    @JoinColumn(name = "arrecadacao_id")
    private Arrecadacao arrecadacaoReferente;

    // Avaliacao 0..1 -> 0..* Notificação
    @OneToMany(mappedBy = "avaliacaoGeradora")
```
